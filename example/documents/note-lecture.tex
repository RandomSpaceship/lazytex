\section{Norms}
A norm on $\R^n$ is a function $\norm{\cdot}$ with these properties.
% enumerate, not itemize: the itemize bullet can need cm-super.
\begin{enumerate}
  \item $\norm{x} \ge 0$, and $\norm{x} = 0$ only if $x = 0$.
  \item $\norm{\lambda x} = |\lambda| \norm{x}$.
  \item $\norm{x + y} \le \norm{x} + \norm{y}$.
\end{enumerate}
\begin{align}
  \norm{x}_2 &= \sqrt{\sum_{i=1}^{n} x_i^2} \\
  \norm{x}_\infty &= \max_i |x_i|
\end{align}

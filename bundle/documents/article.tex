\begin{theorem}
For all $n \ge 1$,
\begin{align}
  \sum_{i=1}^{n} i &= \frac{n(n+1)}{2}, \\
  \int_0^1 f(t)\,\mathrm{d}t &\coloneqq \lim_{k\to\infty} \frac{1}{k} \sum_{j=1}^{k} f\!\left(\tfrac{j}{k}\right).
\end{align}
\end{theorem}
\begin{proof}
By induction, with $\begin{pmatrix} 1 & 2 \\ 3 & 4 \end{pmatrix}$ and $\binom{n}{k}$.
\end{proof}

\pgfdeclareshape{anchored cylinder}{
	\inheritsavedanchors[from=cylinder]
	\inheritanchorborder[from=cylinder]
	\inheritbackgroundpath[from=cylinder]

	% Inherit all the standard cylinder anchors
	\inheritanchor[from=cylinder]{center}
	\inheritanchor[from=cylinder]{north}
	\inheritanchor[from=cylinder]{south}
	\inheritanchor[from=cylinder]{east}
	\inheritanchor[from=cylinder]{west}

	\inheritanchor[from=cylinder]{before top}
	\inheritanchor[from=cylinder]{after top}
	%\anchor{top middle}{%
	\anchor{top}{%
		\csname pgf@anchor@cylinder@before top\endcsname
		\pgf@xa=\pgf@x
		\pgf@ya=\pgf@y
		\csname pgf@anchor@cylinder@after top\endcsname
		\advance\pgf@x by \pgf@xa
		\advance\pgf@y by \pgf@ya
		\pgf@x=.5\pgf@x
		\pgf@y=.5\pgf@y
	}%

	\inheritanchor[from=cylinder]{before bottom}
	\inheritanchor[from=cylinder]{after bottom}
	\anchor{bottom}{%
		\csname pgf@anchor@cylinder@before bottom\endcsname
		\pgf@xa=\pgf@x
		\pgf@ya=\pgf@y
		\csname pgf@anchor@cylinder@after bottom\endcsname
		\advance\pgf@x by \pgf@xa
		\advance\pgf@y by \pgf@ya
		\pgf@x=.5\pgf@x
		\pgf@y=.5\pgf@y
	}%
}
